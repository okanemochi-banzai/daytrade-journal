name: Process trade data

on:
  push:
    branches: [main]
    paths:
      - 'data/*.csv'
      - 'scripts/fetch-prices.mjs'
      - 'scripts/rename-csvs.mjs'
      - '.github/workflows/fetch-prices.yml'
  workflow_dispatch:

permissions:
  contents: write

jobs:
  process:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4
        with:
          token: ${{ secrets.GITHUB_TOKEN }}

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: '20'

      - name: Rename uploaded CSVs to YYYY-MM-DD.csv
        run: node scripts/rename-csvs.mjs

      - name: Commit renamed CSVs (if any)
        id: rename_commit
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add -A data/
          if git diff --staged --quiet; then
            echo "No CSV renames"
            echo "renamed=false" >> $GITHUB_OUTPUT
          else
            git commit -m "🤖 Auto-rename CSV(s) to YYYY-MM-DD format"
            echo "renamed=true" >> $GITHUB_OUTPUT
          fi

      - name: Fetch price data from Yahoo Finance
        run: node scripts/fetch-prices.mjs

      - name: Commit price data
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add data/prices/
          if git diff --staged --quiet; then
            echo "No new price data"
          else
            git commit -m "🤖 Auto-update price data"
          fi

      - name: Push all changes
        run: git push
